}


def ensure_data_file() -> None:
    DATA_DIR.mkdir(parents=True, exist_ok=True)
    if not ENTRIES_PATH.exists():
        ENTRIES_PATH.write_text("[]", encoding="utf-8")


def load_entries() -> list:
    ensure_data_file()
    try:
        return json.loads(ENTRIES_PATH.read_text(encoding="utf-8") or "[]")
    except json.JSONDecodeError:
        return []


def save_entries(entries: list) -> None:
    ensure_data_file()
    ENTRIES_PATH.write_text(json.dumps(entries, indent=2), encoding="utf-8")


def all_players() -> list:
    return [player for box in BOXES for player in box["players"]]


def find_player(player_id: str):
    for player in all_players():
        if player["id"] == player_id:
            return player
    return None


def refresh_points_if_due() -> None:
    """Applies a simple weekly update trigger for a Monday 7:00 PM ET refresh."""
    now = datetime.now(ZoneInfo("America/New_York"))
    current_week_key = now.strftime("%Y-%W")
    state_path = DATA_DIR / "refresh_state.json"
    state = {}
    if state_path.exists():
        try:
            state = json.loads(state_path.read_text(encoding="utf-8") or "{}")
        except json.JSONDecodeError:
            state = {}

    if now.weekday() == 0 and now.hour == 19 and state.get("week_key") != current_week_key:
        state["week_key"] = current_week_key
        state_path.write_text(json.dumps(state, indent=2), encoding="utf-8")


def build_leaderboard(entries: list) -> list:
    refresh_points_if_due()
    rows = []
    for entry in entries:
        total = 0
        for player_id in entry.get("selections", []):
            player = find_player(player_id)
            if player:
                total += int(player.get("points", 0))
        rows.append({
            "name": entry.get("name"),
            "email": entry.get("email"),
            "createdAt": entry.get("createdAt"),
            "total": total,
        })
    rows.sort(key=lambda item: item["total"], reverse=True)
    return rows


@app.route("/")
def index():
    return render_template("index.html")


@app.route("/api/boxes")
def boxes():
    return jsonify({"boxes": BOXES, "prizeInfo": PRIZE_INFO, "payment": PAYMENT_CONFIG})


@app.route("/api/config")
def config():
    return jsonify({"prizeInfo": PRIZE_INFO, "payment": PAYMENT_CONFIG})


@app.route("/api/leaderboard")
def leaderboard():
    entries = load_entries()
    return jsonify({"leaderboard": build_leaderboard(entries)})


@app.route("/api/entry", methods=["POST"])
def submit_entry():
    payload = request.get_json(silent=True) or {}
    name = (payload.get("name") or "").strip()
    email = (payload.get("email") or "").strip()
    selections = payload.get("selections") or []

    if not name or not email or not isinstance(selections, list) or len(selections) != 20:
        return jsonify({"error": "Please provide your name, email, and exactly 20 selections."}), 400

    if len(set(selections)) != 20:
        return jsonify({"error": "Each box must have a unique player selection."}), 400

    for player_id in selections:
        if not find_player(player_id):
            return jsonify({"error": "One or more selections are invalid."}), 400

    entries = load_entries()
    entry = {
        "id": f"entry-{len(entries) + 1}",
        "name": name,
        "email": email,
        "selections": selections,
        "createdAt": datetime.now(ZoneInfo("UTC")).isoformat(),
    }
    entries.append(entry)
    save_entries(entries)
    return jsonify({"message": "Draft entry created successfully.", "leaderboard": build_leaderboard(entries)})
